صفحة الحجز بالموقع

النموذج يجيب الأوقات المتاحة، يرسل الحجز، ويحوّل العميل لصفحة الدفع:

<!-- index.html on GitHub Pages — the booking form posts to your createBooking function -->
<form id="book" dir="rtl">
  <select name="service"><option>تنظيف</option><option>تبييض</option><option>فحص</option></select>
  <input name="name" placeholder="الاسم" required>
  <input name="phone" placeholder="رقم الواتساب" required>
  <input name="email" type="email" placeholder="الإيميل (اختياري)">
  <input name="date" type="date" required> <select name="time" id="slots"></select>
  <button>احجز وادفع</button>
</form>
<script>
const FN = "https://us-central1-YOUR-PROJECT.cloudfunctions.net";
document.querySelector('[name=date]').addEventListener('change', async e => {
  const r = await fetch(`${FN}/freeSlots?date=${e.target.value}`); const free = await r.json();
  slots.innerHTML = free.map(t => `<option>${t}</option>`).join('');
});
book.addEventListener('submit', async e => {
  e.preventDefault();
  const r = await fetch(`${FN}/createBooking`, { method: 'POST', headers: {'Content-Type':'application/json'},
    body: JSON.stringify(Object.fromEntries(new FormData(book))) });
  const { payUrl } = await r.json(); location.href = payUrl;      // Tap / MyFatoorah checkout
});
</script>

الإيميل وقاعدة بيانات العملاء

دالة تشتغل لما يتسجل حجز: تحدّث سجل العميل، ترسل تأكيد له، وإشعار لك:

// functions/onBooking.js — fires when a booking doc is created (Level 3)
const { onDocumentCreated } = require("firebase-functions/v2/firestore");
const { defineSecret } = require("firebase-functions/params");
const admin = require("firebase-admin");
if (!admin.apps.length) admin.initializeApp();
const db = admin.firestore();
const RESEND_KEY = defineSecret("RESEND_KEY");          // resend.com — 3,000 emails/month free
const OWNER = "owner@yourclinic.com";

async function email(to, subject, html) {
  await fetch("https://api.resend.com/emails", { method: "POST",
    headers: { Authorization: `Bearer ${RESEND_KEY.value()}`, "Content-Type": "application/json" },
    body: JSON.stringify({ from: "عيادة النور <book@yourclinic.com>", to, subject, html }) });
}

exports.onBooking = onDocumentCreated({ document: "bookings/{id}", secrets: [RESEND_KEY] }, async (ev) => {
  const b = ev.data.data(), id = ev.params.id;
  // 1) one client record per phone — the "database for all client info"
  const client = db.collection("clients").doc(b.phone);
  await client.set({ name: b.name, phone: b.phone, email: b.email || null,
    lastBooking: b.date, visits: admin.firestore.FieldValue.increment(1), updated: new Date() }, { merge: true });
  // 2) email the customer + notify the owner
  if (b.email) await email(b.email, `تأكيد حجزك — ${b.service} ${b.date} ${b.time}`,
    `<div dir="rtl" style="font-family:sans-serif"><h2>تم استلام حجزك ✅</h2><p>${b.service} — ${b.date} الساعة ${b.time}</p><p>أكمل الدفع هنا: <a href="${b.payUrl}">رابط الدفع</a></p></div>`);
  await email(OWNER, `حجز جديد: ${b.name} — ${b.service}`, `<p>${b.name} · ${b.phone} · ${b.date} ${b.time} · #${id}</p>`);
});

الدفع ← القيد ← الفاتورة
  • createBooking ينشئ charge بـ Tap (درس الحجوزات) ويرجّع payUrl. البوابة تنادي payWebhook (درس المحاسبة) ← bookings/{id}.status = paid + قيد بثلاثة أسطر + PDF.
  • ضف بـ payWebhook: metadata.booking ← حدّث الحجز، ثم أرسل «تم الدفع ✅» على واتساب (sendText من درس واتساب).

الأمان واللوحة

الموقع العام ما يكتب بقاعدة البيانات أبداً — الدوال بس. ولوحتك تحتاج تسجيل دخول (Firebase Auth):

// firestore.rules — the website writes nothing directly; only your functions do
rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    match /{document=**} { allow read, write: if false; }        // public site: no direct access
    match /bookings/{id} { allow read: if request.auth != null; } // your dashboard (owner login)
    match /clients/{id}  { allow read: if request.auth != null; }
    match /journal/{id}  { allow read: if request.auth != null; }
  }
}
  • لوحة /admin (نفس الريبو، صفحة ثانية): مواعيد اليوم، عملاء جدد، دخل الشهر، آخر الفواتير. تقرأ bookings وjournal بعد تسجيل الدخول.
  • نشر: firebase deploy --only functions ثم ارفع ملفات الموقع للريبو. الدومين من إعدادات GitHub Pages.
