الأسرار والدالة

Firebase Functions v2 على Node 20. الأسرار أول:

firebase functions:secrets:set WA_TOKEN
firebase functions:secrets:set WA_VERIFY_TOKEN
firebase functions:secrets:set WA_APP_SECRET
firebase functions:secrets:set WA_PHONE_ID
firebase functions:secrets:set AI_API_KEY

الدالة: تحقق (GET)، توقيع HMAC، ذاكرة محادثة بـ Firestore، نداء الذكاء الاصطناعي، وتحويل لموظف:

// functions/waWebhook.js
const { onRequest } = require("firebase-functions/v2/https");
const { defineSecret } = require("firebase-functions/params");
const admin = require("firebase-admin");
const crypto = require("crypto");
if (!admin.apps.length) admin.initializeApp();
const db = admin.firestore();

const WA_TOKEN = defineSecret("WA_TOKEN"), WA_VERIFY_TOKEN = defineSecret("WA_VERIFY_TOKEN"),
      WA_APP_SECRET = defineSecret("WA_APP_SECRET"), WA_PHONE_ID = defineSecret("WA_PHONE_ID"),
      AI_API_KEY = defineSecret("AI_API_KEY");
const GRAPH = "https://graph.facebook.com/v21.0"; // use the version shown in your Meta dashboard

const FACTS = `الاسم: [اسم الشركة]. الدوام: [الدوام]. الموقع: [الموقع]. الأسعار: [الأسعار].`;
const SYSTEM = `أنت مساعد واتساب لشركة بالكويت. رد بلهجة كويتية مختصرة (استخدم "منو" مو "مين")،
أقصى ٣ أسطر. جاوب فقط من هالمعلومات: ${FACTS}
إذا السؤال خارج المعلومات، أو العميل يبي يحجز أو يكلم موظف، رد بالضبط: HANDOFF`;

async function sendText(to, body) {
  const r = await fetch(`${GRAPH}/${WA_PHONE_ID.value()}/messages`, {
    method: "POST",
    headers: { Authorization: `Bearer ${WA_TOKEN.value()}`, "Content-Type": "application/json" },
    body: JSON.stringify({ messaging_product: "whatsapp", to, type: "text", text: { body } }),
  });
  if (!r.ok) console.error("send", r.status, await r.text());
}

async function askAI(history) {
  const r = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "x-api-key": AI_API_KEY.value(), "anthropic-version": "2023-06-01", "Content-Type": "application/json" },
    body: JSON.stringify({ model: "claude-haiku-4-5", max_tokens: 300, system: SYSTEM, messages: history }),
  });
  const j = await r.json();
  return (j.content || []).map(c => c.text || "").join("").trim();
}

exports.waWebhook = onRequest(
  { region: "us-central1", secrets: [WA_TOKEN, WA_VERIFY_TOKEN, WA_APP_SECRET, WA_PHONE_ID, AI_API_KEY] },
  async (req, res) => {
    if (req.method === "GET") {
      const ok = req.query["hub.mode"] === "subscribe" && req.query["hub.verify_token"] === WA_VERIFY_TOKEN.value();
      return ok ? res.status(200).send(req.query["hub.challenge"]) : res.sendStatus(403);
    }
    const sig = req.get("x-hub-signature-256") || "";
    const exp = "sha256=" + crypto.createHmac("sha256", WA_APP_SECRET.value()).update(req.rawBody).digest("hex");
    if (sig.length !== exp.length || !crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(exp))) return res.sendStatus(401);
    res.sendStatus(200); // ack within 20s or Meta retries

    for (const e of req.body.entry || []) for (const ch of e.changes || []) {
      const msg = ch.value?.messages?.[0];
      if (!msg || msg.type !== "text") continue;
      const from = msg.from, text = msg.text.body.trim();

      const ref = db.collection("wa_chats").doc(from);
      const snap = await ref.get();
      const d = snap.exists ? snap.data() : { history: [], human: false, seen: [] };
      if (d.seen?.includes(msg.id)) continue;   // Meta retries
      if (d.human) continue;                      // a person took over

      const history = [...d.history.slice(-10), { role: "user", content: text }];
      let reply = await askAI(history);
      if (reply === "HANDOFF" || /موظف|انسان|شخص/.test(text)) {
        reply = "تمام، بيتواصل معاك أحد الفريق خلال ساعة 🙏";
        await db.collection("wa_handoffs").add({ from, text, at: new Date() });
        d.human = true;
      }
      await sendText(from, reply);
      await ref.set({ history: [...history, { role: "assistant", content: reply }], human: d.human,
                      seen: [...(d.seen || []).slice(-50), msg.id], updated: new Date() }, { merge: true });
    }
  }
);

القواعد اللي لازم تعرفها
  • ترد بحرية خلال ٢٤ ساعة من آخر رسالة من العميل. بعدها تحتاج Template معتمد (مناسب لـ «تم تأكيد حجزك»، مو للدردشة).
  • أول ١٠٠٠ محادثة خدمة بالشهر مجانية، بعدها Meta تحاسب على المحادثة. تكلفة الذكاء الاصطناعي بنموذج صغير أجزاء من الفلس للرسالة.
  • خلّ FACTS معلومات تحطها بموقعك. لا تخليه يخترع أسعار — «HANDOFF» أرخص من رقم غلط.
