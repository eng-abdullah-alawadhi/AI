الدوال: المتاح، الدفع، التأكيد، التذكير

Firebase Functions v2، Node 20. السر: firebase functions:secrets:set TAP_SECRET. ثلاث دوال HTTP ودالة مجدولة:

// functions/booking.js — slot check + Tap payment + confirmation + reminder
const { onRequest } = require("firebase-functions/v2/https");
const { onSchedule } = require("firebase-functions/v2/scheduler");
const { defineSecret } = require("firebase-functions/params");
const admin = require("firebase-admin");
if (!admin.apps.length) admin.initializeApp();
const db = admin.firestore();
const TAP_SECRET = defineSecret("TAP_SECRET");

// 1) Free slots for a day: working hours minus booked ones
const HOURS = ["10:00", "11:30", "13:30", "16:00", "17:30"]; // your slots
async function freeSlots(dateISO) {
  const booked = await db.collection("bookings")
    .where("date", "==", dateISO).where("status", "in", ["paid", "pending"]).get();
  const taken = new Set(booked.docs.map(d => d.data().time));
  return HOURS.filter(h => !taken.has(h));
}

// 2) Called by your WhatsApp/Instagram bot (Level 3 of the previous guides)
//    POST { phone, name, date, time }  → returns a Tap payment link
exports.createBooking = onRequest({ region: "us-central1", secrets: [TAP_SECRET] }, async (req, res) => {
  const { phone, name, date, time } = req.body || {};
  if (!(await freeSlots(date)).includes(time)) return res.status(409).json({ error: "slot_taken", free: await freeSlots(date) });
  const ref = await db.collection("bookings").add({ phone, name, date, time, status: "pending", created: new Date() });

  const r = await fetch("https://api.tap.company/v2/charges", {
    method: "POST",
    headers: { Authorization: `Bearer ${TAP_SECRET.value()}`, "Content-Type": "application/json" },
    body: JSON.stringify({
      amount: 25, currency: "KWD",                     // your price
      customer: { first_name: name, phone: { country_code: "965", number: phone.replace(/^965/, "") } },
      source: { id: "src_all" },
      redirect: { url: `https://YOUR-SITE/thanks?b=${ref.id}` },
      post: { url: "https://us-central1-YOUR-PROJECT.cloudfunctions.net/tapWebhook" },
      metadata: { booking: ref.id },
    }),
  });
  const charge = await r.json();
  await ref.update({ tapCharge: charge.id });
  res.json({ bookingId: ref.id, payUrl: charge.transaction.url });
});

// 3) Tap tells us when it's paid → confirm
exports.tapWebhook = onRequest({ region: "us-central1" }, async (req, res) => {
  const c = req.body;
  if (c.status === "CAPTURED" && c.metadata?.booking) {
    await db.collection("bookings").doc(c.metadata.booking).update({ status: "paid", paidAt: new Date() });
    // sendWhatsApp(phone, `تم الدفع ✅ موعدك ${date} ${time} — بنذكّرك قبلها بساعتين`)  ← reuse sendText() from the WhatsApp guide
  }
  res.sendStatus(200);
});

// 4) Reminder 2 hours before, every 15 minutes
exports.reminders = onSchedule({ schedule: "every 15 minutes", region: "us-central1", timeZone: "Asia/Kuwait" }, async () => {
  const now = new Date(); const soon = new Date(now.getTime() + 2 * 3600e3);
  const snap = await db.collection("bookings").where("status", "==", "paid").where("reminded", "==", false).get();
  for (const d of snap.docs) {
    const b = d.data(); const at = new Date(`${b.date}T${b.time}:00+03:00`);
    if (at > now && at <= soon) {
      // sendWhatsApp(b.phone, `تذكير 🔔 موعدك اليوم ${b.time}`)
      await d.ref.update({ reminded: true });
    }
  }
});

firebase deploy --only functions:createBooking,functions:tapWebhook,functions:reminders

اربط البوت بالدوال
  • بحالة البوت (wa_chats) أضف stage: day | time | paying وخزّن اليوم والوقت المختارين.
  • لما يختار الوقت: استدعِ createBooking وأرسل payUrl للعميل.
  • لو رجّعت slot_taken: اعرض الأوقات الجديدة من free واطلب يختار مرة ثانية.
  • tapWebhook يأكّد ويرسل رسالة «تم الدفع ✅» — البوت ما يحتاج يسوي شي.
4

Tap Payments
  • لوحة Tap ← Developers ← Secret key (test أول، بعدين live). KNET يشتغل بـ src_all أو src_kw.knet.
  • الـ post.url هو اللي يخلي Tap يبلّغ دالتك لما ينتهي الدفع. تحقق من status === "CAPTURED" فقط.
  • الحجوزات pending اللي ما اندفعت خلال ٣٠ دقيقة: دالة مجدولة تحولها cancelled عشان الوقت يرجع متاح.
5

لوحة للمالك
  • صفحة بسيطة بلوحة التحكم: حجوزات اليوم، المدفوع، الملغي، وزر «إلغاء» يحدّث الحالة ويرسل رسالة للعميل.
  • تقرير صباحي على واتساب للمالك: «اليوم ٦ مواعيد، ٤ مدفوعة» — هذا موضوع البوست الجاي.
