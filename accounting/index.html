ويب هوك واحد لكل البوابات

Firebase Functions v2، Node 20. دالة وحدة تستقبل Tap وUPayments وMyFatoorah، توحّد الشكل، وتسجّل القيد بثلاثة أسطر:

// functions/payWebhook.js — one webhook, three gateways → one ledger
const { onRequest } = require("firebase-functions/v2/https");
const admin = require("firebase-admin");
if (!admin.apps.length) admin.initializeApp();
const db = admin.firestore();
const FEE = { tap: 0.025, upayments: 0.025, myfatoorah: 0.02, knet: 0.015 }; // your contract rates

// Normalise each gateway's payload to { ref, amount, gateway, customer, paid }
function parse(gw, b) {
  if (gw === "tap")        return { ref: b.id,            amount: +b.amount,         customer: b.customer?.first_name, paid: b.status === "CAPTURED" };
  if (gw === "upayments")  return { ref: b.payment_id,    amount: +b.amount,         customer: b.customer_name,        paid: b.result === "CAPTURED" };
  if (gw === "myfatoorah") return { ref: b.InvoiceId,     amount: +b.InvoiceValue,   customer: b.CustomerName,         paid: b.TransactionStatus === "SUCCESS" };
}

exports.payWebhook = onRequest({ region: "us-central1" }, async (req, res) => {
  const gw = (req.query.gw || "").toLowerCase();           // …/payWebhook?gw=tap
  const p = parse(gw, req.body); if (!p) return res.sendStatus(400);
  res.sendStatus(200);                                     // ack fast
  if (!p.paid) return;
  const ref = db.collection("payments").doc(`${gw}_${p.ref}`);
  if ((await ref.get()).exists) return;                    // idempotent: gateways retry
  const fee = +(p.amount * FEE[gw]).toFixed(3), net = +(p.amount - fee).toFixed(3);
  const batch = db.batch(); const at = new Date();
  batch.set(ref, { ...p, gateway: gw, fee, net, at });
  const je = db.collection("journal").doc();               // one journal entry, three lines
  batch.set(je, { at, ref: ref.id, lines: [
    { account: "4000-Revenue",       credit: p.amount },
    { account: "6100-Gateway fees",  debit:  fee },
    { account: "1010-Bank",          debit:  net },
  ]});
  await batch.commit();
  // await makeInvoicePdf(ref.id) then sendText(customerPhone, "🧾 فاتورتك …")  ← see the WhatsApp guide
});
firebase deploy --only functions:payWebhook

سجّل الرابط عند كل بوابة
  • Tap: Developers ← Webhooks ← …/payWebhook?gw=tap. أو مرّره بكل charge بـ post.url.
  • UPayments: Settings ← Webhook URL ← …?gw=upayments. MyFatoorah: Webhooks ← …?gw=myfatoorah ← فعّل «TransactionsStatusChanged».
  • تحقق من التوقيع لكل بوابة (كل وحدة لها طريقتها) قبل ما تثق بالطلب — نفس فكرة HMAC بدرس واتساب.

الفاتورة PDF
  • دالة makeInvoicePdf: قالب HTML بالعربي (نفس خطوط موقعك) ← Puppeteer/Chromium ← PDF ← Cloud Storage ← رابط موقّع.
  • أرسل الرابط بـ sendText من درس واتساب، أو كمستند عبر Cloud API (type: "document").

اللوحة والتقرير
  • صفحة بلوحتك تقرأ journal: مبيعات اليوم، الرسوم، الصافي، وآخر ١٠ عمليات. (التجربة الحيّة فيها نفس الشكل.)
  • دالة مجدولة أول كل شهر ← CSV للمحاسب بأعمدة: التاريخ، الحساب، مدين، دائن، المرجع.
  • لا تحسب الضريبة بنفسك قبل ما تسأل محاسبك — القواعد تختلف حسب النشاط.
